void MainWindow::runAutoBenchDemo(unsigned cyclicSeconds) {
    refreshAdapterList();
    if (ioPerformanceChart_ != nullptr) {
        ioPerformanceChart_->clearTrend();

    QTimer::singleShot(
        800,
        this,
        [this, cyclicSeconds]() {
            setStatusText(
                QStringLiteral("Auto-bench: running I/O EXCHANGE bench"));
            lastRunScenarioId_ = QStringLiteral("io_exchange_bench");
            if (ioBenchDurationSpin_ != nullptr) {
                ioBenchDurationSpin_->setValue(static_cast<int>(cyclicSeconds));
            }
            if (ioBenchPeriodSpin_ != nullptr) {
                ioBenchPeriodSpin_->setValue(0);
            }
            onRunIoBench();
        });
}

    const int ioTab = tabIndexByLabel(QStringLiteral("I/O Performance"));

        QStringLiteral("io_exchange_bench"),
        {},
        selectedAdapterPath(),
        selectedAdapterLabel(),
        peerMacEdit_->text(),
        duration,
        period);
}

int MainWindow::tabIndexByLabel(const QString& label) const {
    if (mainTabs_ == nullptr) {
        return 0;
    }
    for (int i = 0; i < mainTabs_->count(); ++i) {
        if (mainTabs_->tabText(i) == label) {
            return i;
        }
    }
    return 0;
}

void MainWindow::populateIoPerformanceStats(const LeapConformanceMetrics& metrics) {
    if (ioBenchStatsTable_ == nullptr) {
        return;
    }

    const auto set = [this](int row, const QString& key, const QString& val) {
        ioBenchStatsTable_->setItem(row, 0, new QTableWidgetItem(key));
        ioBenchStatsTable_->setItem(row, 1, new QTableWidgetItem(val));
    };

    uint64_t avgRtt = 0u;
    if (metrics.pd.network_rtt_samples > 0u) {
        avgRtt = metrics.pd.total_network_rtt_us / metrics.pd.network_rtt_samples;
    }

    double exchPerSec = 0.0;
    if (ioBenchDurationSpin_ != nullptr && ioBenchDurationSpin_->value() > 0) {
        exchPerSec = static_cast<double>(metrics.pd.exchange_replies) /
                     static_cast<double>(ioBenchDurationSpin_->value());
    }

    set(0, QStringLiteral("wire_rtt_last"),
        metrics.pd.network_rtt_samples > 0u
            ? QStringLiteral("%1 µs").arg(metrics.pd.last_network_rtt_us)
            : QStringLiteral("—"));
    set(1, QStringLiteral("wire_rtt_avg"),
        metrics.pd.network_rtt_samples > 0u
            ? QStringLiteral("%1 µs").arg(avgRtt)
            : QStringLiteral("—"));
    set(2, QStringLiteral("wire_rtt_max"),
        metrics.pd.network_rtt_samples > 0u
            ? QStringLiteral("%1 µs").arg(metrics.pd.max_network_rtt_us)
            : QStringLiteral("—"));
    set(3, QStringLiteral("exchange_replies"),
        QString::number(metrics.pd.exchange_replies));
    set(4, QStringLiteral("cycles_completed"),
        QString::number(metrics.pd.cycles_completed));
    set(5, QStringLiteral("recv_timeouts"),
        QString::number(metrics.pd.recv_timeouts));
    set(6, QStringLiteral("lost_frames"),
        QString::number(metrics.pd.lost_frames));
    set(7, QStringLiteral("exchanges_per_sec"),
        metrics.pd.exchange_replies > 0u
            ? QString::number(exchPerSec, 'f', 1)
            : QStringLiteral("—"));

    if (ioPerformanceChart_ != nullptr) {
        ioPerformanceChart_->applyMetrics(metrics);
    }
}
